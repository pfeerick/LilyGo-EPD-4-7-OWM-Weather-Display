cmake_minimum_required(VERSION 3.20)
project(epd_native_tests CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)

if(MSVC)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
  add_compile_options(/W3 /wd4244 /wd4267 /wd4305)
else()
  add_compile_options(-Wall -Wno-unused-function -Wno-sign-compare -Wno-narrowing)
endif()
if(WIN32)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()

set(ROOT "${CMAKE_SOURCE_DIR}/..")
set(SIM "${ROOT}/simulator")

if(NOT EXISTS "${ROOT}/.pio/libdeps/display/ArduinoJson/src/ArduinoJson.h")
  message(FATAL_ERROR "ArduinoJson not found — run `pio pkg install` first to fetch library deps")
endif()

# src/setup_screen.cpp includes build_info.h, normally generated by the PlatformIO pre-build step.
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND ${Python3_EXECUTABLE} ${ROOT}/scripts/gen_build_info.py ${ROOT})

# doctest (header-only). SOURCE_SUBDIR points at a non-existent dir so only the header is fetched and
# doctest's own CMakeLists (which declares an old cmake_minimum_required rejected by CMake 4) is skipped.
include(FetchContent)
FetchContent_Declare(
  doctest
  URL https://github.com/doctest/doctest/archive/refs/tags/v2.4.11.tar.gz
  URL_HASH SHA256=632ed2c05a7f53fa961381497bf8069093f0d6628c5f26286161fbd32a560186
  SOURCE_SUBDIR header-only
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
FetchContent_MakeAvailable(doctest)

# The code under test is compiled as part of test_main.cpp (via simulator/main_wasm.cpp), so file-static
# helpers in src/ are reachable from the tests. Only the simulator's C support code is built separately.
add_executable(native_tests
  test_main.cpp
  ${SIM}/epdiy_pc/epdiy.c
  ${SIM}/epdiy_pc/font.c
  ${SIM}/miniz/miniz.c
  ${SIM}/miniz/miniz_tdef.c
  ${SIM}/miniz/miniz_tinfl.c
  ${SIM}/miniz/miniz_zip.c
  ${SIM}/qrcodegen/qrcodegen.c
)

target_include_directories(native_tests PRIVATE
  ${CMAKE_SOURCE_DIR}
  ${SIM}
  ${SIM}/esp_stubs
  ${SIM}/miniz
  ${SIM}/qrcodegen
  ${ROOT}/include
  ${ROOT}/src
  ${ROOT}/.pio/libdeps/display/epdiy/src
  ${ROOT}/.pio/libdeps/display/ArduinoJson/src
  ${doctest_SOURCE_DIR}
)

target_compile_definitions(native_tests PRIVATE SIMULATOR_BUILD=1)

enable_testing()
add_test(NAME native_tests COMMAND native_tests)
